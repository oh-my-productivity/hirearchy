usage: orgmcp define-service [-h] [--description DESCRIPTION] [--input 'name*:description']
                             [--acceptance TEXT|@FILE] [--add-acceptance TEXT|@FILE]
                             [--remove-acceptance TEXT|@FILE] [--deliverable DELIVERABLE]
                             [--format FORMAT] [--max-words MAX_WORDS] [--path PATH]
                             [--budget BUDGET] [--playbook @FILE] [--locked | --no-locked]
                             [--notes NOTES]
                             owner name

positional arguments:
  owner                 the top-level agent that offers the service
  name                  lowercase-dashed service name

options:
  -h, --help            show this help message and exit
  --description DESCRIPTION
                        one line: what the service delivers (a first definition needs it)
  --input 'name*:description'
                        an input of a request; a trailing * on the name = required (repeatable)
  --acceptance TEXT|@FILE
                        a criterion, or @file with one criterion per line (repeatable); replaces
                        all existing ones
  --add-acceptance TEXT|@FILE
                        amendment: add a criterion (repeatable); not with --acceptance
  --remove-acceptance TEXT|@FILE
                        amendment: drop a criterion, by its exact text (repeatable); not with
                        --acceptance
  --deliverable DELIVERABLE
                        what is handed back (a first definition needs it)
  --format FORMAT
  --max-words MAX_WORDS
  --path PATH           absolute folder for the deliverable
  --budget BUDGET       runs handed down with each request (a first definition needs it)
  --playbook @FILE      initial playbook (first definition only)
  --locked, --no-locked
                        --locked: the owner may not revise the playbook; --no-locked undoes it
  --notes NOTES         free text for the owner, or @file
