usage: hirearchy request [-h] [--input KEY=VALUE] [--title TITLE] [--note NOTE] [--budget BUDGET]
                         [--revises TASK_ID]
                         service

positional arguments:
  service

options:
  -h, --help         show this help message and exit
  --input KEY=VALUE  a value for an input of the service (repeatable)
  --title TITLE
  --note NOTE        free text for the owner
  --budget BUDGET    runs handed down (default: the contract's)
  --revises TASK_ID  send back an earlier request of this service for rework
